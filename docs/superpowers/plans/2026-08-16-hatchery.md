# 부화소 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 인벤토리에 쌓이기만 하던 알을 부화소에서 깨서 헛간에 새 케어를 얻는다.

**Architecture:** 규칙은 순수 C# `HatchService.TryHatch` 한 곳에 두고, `HatchPanel`(`GamePanel` 상속)은 표시와 배선만 한다. 상점(`ShopService` + `ShopPanel`)과 같은 구조다. 세이브 포맷은 v3 그대로다 — 부화가 바꾸는 인벤토리와 로스터를 v3 가 이미 저장한다.

**Tech Stack:** Unity 6000.2.8f1 (2D), C#, NUnit(EditMode), uGUI. 씬은 `GameBootstrap.BuildAll()` 이 전량 생성한다.

**Spec:** `docs/superpowers/specs/2026-08-16-hatchery-design.md`

## Global Constraints

- **씬을 손으로 편집하지 않는다.** `GameBootstrap.cs` 를 고치고 `BuildAll` 로 재생성한다.
- **UI 행·버튼을 런타임에 생성하지 않는다.** 개수는 씬 생성 시점에 고정된다 (함정 3.1).
- **uGUI persistent listener 는 무인자 또는 int 인자만 쓴다** (함정 3.6). Object 인자는 직렬화되지 않는다.
- **패널은 배선이 끝난 뒤에 닫아서 저장한다** (함정 3.8).
- **데이터에서 개수가 나오는 UI 는 0 을 `Debug.LogError` 로 크게 실패시킨다** (함정 3.9/3.10).
- **배치모드 실행은 Unity 에디터가 닫혀 있어야 한다** (함정 3.5). 열려 있으면 프로젝트 잠금으로 실패한다.
- **종료 코드 0 을 믿지 않는다.** 생성물(씬·`results.xml`)을 직접 확인한다.
- Core 계층(`Assets/Scripts/Runtime/Core/`)은 `UnityEngine` UI/MonoBehaviour 에 의존하지 않는다. `ScriptableObject` 인자는 허용된다 (`CareActions.Equip(ItemData)` 선례).
- 표시 문자열은 전부 `PanelLabels` 에 모은다 — 그래야 `PanelLabelsTests` 가 덮는다.
- 브랜치: `feature/hatchery` (이미 만들어져 있고 설계 문서가 커밋돼 있다).

using FarmCreature.Core;
using FarmCreature.Data;
using NUnit.Framework;

namespace FarmCreature.Tests
{
    public class HatchServiceTests
    {
        private CreatureSpeciesData sprigling;
        private EggData egg;

        [SetUp]
        public void SetUp()
        {
            sprigling = TestFixtures.MakeSpecies("sprigling", "스프리글링", 5, 4, 20);
            egg = TestFixtures.MakeEgg("egg_sprigling", "스프리글링 알", sprigling);
        }

        private GameState EmptyState()
        {
            return new GameState(new Roster(), new Inventory(), 0);
        }

        private GameState StateWithEggs(int eggCount)
        {
            var state = EmptyState();
            state.Inventory.Add(egg.eggId, eggCount);
            return state;
        }

        [Test]
        public void Hatch_ConsumesEggAndAddsCreature()
        {
            var state = StateWithEggs(1);

            var result = HatchService.TryHatch(state, egg);

            Assert.IsTrue(result.Success, result.Message);
            Assert.AreEqual(0, state.Inventory.GetCount("egg_sprigling"));
            Assert.AreEqual(1, state.Roster.Count);
            StringAssert.Contains("스프리글링", result.Message);
        }

        [Test]
        public void Hatch_ProducesSpeciesFromEggWithBaseStats()
        {
            var state = StateWithEggs(1);

            HatchService.TryHatch(state, egg);
            var hatched = state.Roster.GetMember(0);

            Assert.AreSame(sprigling, hatched.CurrentSpecies);
            Assert.AreEqual(5, hatched.GetBaseStat(StatType.Attack));
            Assert.AreEqual(4, hatched.GetBaseStat(StatType.Defense));
            Assert.AreEqual(20, hatched.GetBaseStat(StatType.Health));
            Assert.AreEqual(0, hatched.Exp, "갓 난 케어는 경험치가 0 이다");
        }

        [Test]
        public void Hatch_WithoutEgg_ChangesNothing()
        {
            var state = EmptyState();

            var result = HatchService.TryHatch(state, egg);

            Assert.IsFalse(result.Success);
            Assert.AreEqual(0, state.Roster.Count, "실패하면 케어가 늘지 않는다");
            Assert.AreEqual(0, state.Inventory.GetCount("egg_sprigling"));
        }

        [Test]
        public void Hatch_WhenBarnFull_KeepsTheEgg()
        {
            var state = StateWithEggs(1);
            for (int i = 0; i < Roster.MaxMembers; i++)
            {
                state.Roster.Add(CreatureInstance.CreateNew(sprigling));
            }

            var result = HatchService.TryHatch(state, egg);

            Assert.IsFalse(result.Success);
            Assert.AreEqual(Roster.MaxMembers, state.Roster.Count);
            Assert.AreEqual(1, state.Inventory.GetCount("egg_sprigling"),

            StringAssert.Contains("헛간", result.Message);
        }

        [Test]
        public void Hatch_EggWithoutTarget_IsRejected()
        {
            var broken = TestFixtures.MakeEgg("egg_broken", "빈 알");
            var state = EmptyState();
            state.Inventory.Add("egg_broken", 1);

            var result = HatchService.TryHatch(state, broken);

            Assert.IsFalse(result.Success, "무엇이 나오는지 없는 알은 데이터 실수다");

            Assert.AreEqual(0, state.Roster.Count);
        }

        [Test]
        public void Hatch_RejectsNullStateOrEgg()
        {
            var state = StateWithEggs(1);

            Assert.IsFalse(HatchService.TryHatch(null, egg).Success);
            Assert.IsFalse(HatchService.TryHatch(state, null).Success);
            Assert.AreEqual(1, state.Inventory.GetCount("egg_sprigling"));
            Assert.AreEqual(0, state.Roster.Count);
        }

        [Test]
        public void Hatch_Twice_ProducesTwoDistinctCreatures()
        {
            var state = StateWithEggs(2);

            Assert.IsTrue(HatchService.TryHatch(state, egg).Success);
            Assert.IsTrue(HatchService.TryHatch(state, egg).Success);

            Assert.AreEqual(2, state.Roster.Count);
            Assert.AreEqual(0, state.Inventory.GetCount("egg_sprigling"));
            Assert.AreNotSame(state.Roster.GetMember(0), state.Roster.GetMember(1),
                "같은 인스턴스를 두 번 넣으면 Roster.Add 가 거부한다");
        }
    }

```csharp
using FarmCreature.Data;

namespace FarmCreature.Core
{
    /// <summary>
    /// 부화 규칙. <see cref="ShopService"/> 와 같은 자리·같은 성격이다 —
    /// 순수 C# 이라 EditMode 테스트가 에셋 없이 돈다.
    ///
    /// <b>검사를 전부 끝낸 뒤에 알을 소모한다.</b> 부화는 인벤토리와 로스터 두 곳을
    /// 건드리므로, 순서를 뒤집으면 "알은 사라졌는데 케어는 없다" 가 실재한다 (설계 2.1).
    /// 상점은 "차감 먼저, 지급 나중" 으로 같은 보장을 얻었다 — 순서만 반대이고 보장은 같다.
    /// </summary>
    public static class HatchService
    {
        public static ActionResult TryHatch(GameState state, EggData egg)
        {
            if (state == null || state.Inventory == null || state.Roster == null
                || egg == null || string.IsNullOrEmpty(egg.eggId) || egg.hatchesInto == null)
            {
                return ActionResult.Fail("부화시킬 수 없는 알이다.");
            }

            if (state.Inventory.GetCount(egg.eggId) < 1)
            {
                return ActionResult.Fail("가진 알이 없다.");
            }

            // 자리를 '알을 깨기 전에' 본다. 뒤에 보면 알만 사라진다.
            if (state.Roster.IsFull)
            {
                return ActionResult.Fail("헛간이 가득 찼다. (최대 " + Roster.MaxMembers + "마리)");
            }

            state.Inventory.TryConsume(egg.eggId, 1);

            // 위 두 검사를 통과했으므로 Add 는 실패할 수 없다 — 새 인스턴스라 중복이 아니고 자리도 있다.
            var hatched = CreatureInstance.CreateNew(egg.hatchesInto);
            state.Roster.Add(hatched);

            return ActionResult.Ok(EggName(egg) + "에서 " + SpeciesName(egg) + "이(가) 나왔다.");
        }

        private static string EggName(EggData egg)
        {
            return string.IsNullOrEmpty(egg.displayName) ? egg.eggId : egg.displayName;
        }

        private static string SpeciesName(EggData egg)
        {
            return string.IsNullOrEmpty(egg.hatchesInto.displayName)
                ? egg.hatchesInto.speciesId
                : egg.hatchesInto.displayName;
        }
    }
}
```

    /// <summary>
    /// 부화소 — 알을 깨서 새 케어를 얻는다. 행은 <see cref="EggDatabase"/> 인덱스와 1:1 이고
    /// 개수가 씬 생성 시점에 고정된다 (함정 3.1). [부화] 버튼은 <b>int 인자 persistent listener</b>
    /// 로 그 인덱스를 넘긴다 (함정 3.6).
    ///
    /// 규칙은 <see cref="HatchService"/> 한 곳에만 있다 — 이 패널은 표시와 배선만 한다.
    /// </summary>
    public class HatchPanel : GamePanel
    {
        public Text titleText;

        [Tooltip("EggDatabase 인덱스 순서와 같아야 한다. GameBootstrap 이 채운다.")]
        public Text[] rowTexts;

        [Tooltip("알 목록. 행 개수가 여기서 나온다. GameBootstrap 이 채운다.")]
        public EggDatabase eggDatabase;

        [Tooltip("로그와 전체 갱신을 맡길 조율자. GameBootstrap 이 채운다.")]
        public CareManager careManager;

        /// <summary>uGUI 버튼이 알 인덱스를 넘긴다.</summary>
        public void Hatch(int eggIndex)
        {
            var egg = EggAt(eggIndex);
            if (egg == null)
            {
                Log("그런 알은 없다.");
                return;
            }

            var result = HatchService.TryHatch(state, egg);

            // 성공이든 실패든 다시 그린다 — 헛간 보유 줄과 제목의 n/8 이 함께 바뀐다.
            if (careManager != null)
            {
                careManager.RefreshAll();
            }
            else
            {
                Refresh();
            }

            Log(result.Message);
        }

        public override void Refresh()
        {
            if (titleText != null)
            {
                titleText.text = PanelLabels.HatchTitle(state);
            }

            if (rowTexts == null)
            {
                return;
            }

            var inventory = Inventory;

            for (int i = 0; i < rowTexts.Length; i++)
            {
                if (rowTexts[i] == null)
                {
                    continue;
                }

                var egg = EggAt(i);
                if (egg == null)
                {
                    rowTexts[i].text = string.Empty;
                    continue;
                }

                int owned = inventory != null ? inventory.GetCount(egg.eggId) : 0;
                rowTexts[i].text = PanelLabels.HatchRow(egg, owned);
            }
        }

        private EggData EggAt(int index)
        {
            if (eggDatabase == null || eggDatabase.eggs == null
                || index < 0 || index >= eggDatabase.eggs.Count)
            {
                return null;
            }

            return eggDatabase.eggs[index];
        }

        private void Log(string message)
        {
            if (careManager != null)
            {
                careManager.SetLog(message);
            }
        }
    }
}

```csharp
        /// <summary>
        /// 부화소 패널. 행은 <see cref="EggDatabase"/> 인덱스와 1:1 이고
        /// [부화] 버튼이 그 인덱스를 int 인자로 넘긴다 (함정 3.6).
        /// 알이 1종뿐이라 상점(620)보다 낮은 420 이다 — 알이 4종을 넘으면 올려야 한다.
        /// </summary>
        private static HatchPanel BuildHatchPanel(
            Transform parent, CareManager careManager, EggDatabase eggDb)
        {
            var panelGo = CreatePanelBackground(parent, "HatchPanel", new Vector2(900f, 420f));
            var panel = panelGo.AddComponent<HatchPanel>();
            panel.eggDatabase = eggDb;
            panel.careManager = careManager;

            panel.titleText = CreatePanelText(panelGo.transform, "Title", new Vector2(0f, 160f),
                new Vector2(860f, 50f), TextAnchor.MiddleCenter, 34, Color.white);
            panel.titleText.text = "부화소";

            int rowCount = eggDb != null && eggDb.eggs != null ? eggDb.eggs.Count : 0;

            // 0 은 정상 상황이 아니다 — 참조가 죽었다는 뜻이다 (함정 3.9/3.10).
            if (rowCount == 0)
            {
                Debug.LogError("[FarmCreature] 부화소 패널: 알이 0종이다. "
                    + "EggDatabase 참조가 죽었거나 비어 있다. eggDb==null? " + (eggDb == null));
            }

            var rows = new Text[rowCount];
            float rowY = 60f;

            for (int i = 0; i < rowCount; i++)
            {
                rows[i] = CreatePanelText(panelGo.transform, "Row" + i, new Vector2(-150f, rowY),
                    new Vector2(560f, 50f), TextAnchor.MiddleLeft, 26, Color.white);

                var hatchButton = CreatePanelButton(panelGo.transform, "Hatch" + i, "부화",
                    new Vector2(300f, rowY), new Vector2(150f, 54f),
                    new Color(0.55f, 0.45f, 0.28f));
                UnityEventTools.AddIntPersistentListener(
                    hatchButton.onClick, new UnityAction<int>(panel.Hatch), i);

                rowY -= 80f;
            }

            panel.rowTexts = rows;

            var closeButton = CreatePanelButton(panelGo.transform, "Close", "닫기",
                new Vector2(0f, -140f), new Vector2(220f, 60f), new Color(0.4f, 0.4f, 0.46f));
            UnityEventTools.AddVoidPersistentListener(
                closeButton.onClick, new UnityAction(panel.Close));

            return panel;
        }
```
